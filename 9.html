구글 시트 1분 설정 방법
  1. 새 구글 스프레드시트를 열고 확장 프로그램 > Apps Script 클릭
  2. 기존 코드를 지우고 아래 코드를 붙여넣은 뒤 저장
  3. 우측 상단 배포 > 새 배포 > 웹 앱 선택 (액세스 권한: 모든 사용자) 후 나온 URL을 위에 입력
function doGet() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
  const val = sheet.getRange('A1').getValue();
  return ContentService.createTextOutput(val || '{}')
    .setMimeType(ContentService.MimeType.JSON);
}
function doPost(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
  sheet.getRange('A1').setValue(e.postData.contents);
  sheet.getRange('A2').setValue('최근 업데이트: ' + new Date().toLocaleString('ko-KR'));
  return ContentService.createTextOutput(JSON.stringify({ ok: true }))
    .setMimeType(ContentService.MimeType.JSON);
}
